2. Display information of all students

3. Display students belonging to Computer department
db.student.find({
    dept: "Computer"
})

4. Display students belonging to Computer or IT department using logical operator
db.student.find({
    $or: [
        { dept: "Computer" },
        { dept: "IT" }
    ]
})

5. Display students whose marks are greater than 70
db.student.find({
    marks: { $gt: 70 }
})

6. Display students whose marks are greater than or equal to 80 and belong to Computer department
db.student.find({
    marks: { $gte: 80 },
    dept: "Computer"
})

7. Display students whose marks are less than 40 or belong to IT department
db.student.find({
    $or: [
        { marks: { $lt: 40 } },
        { dept: "IT" }
    ]
})

8. Display students whose marks are between 40 and 80
db.student.find({
    marks: {
        $gte: 40,
        $lte: 80
    }
})

9. Display students whose department is not IT
db.student.find({
    dept: { $ne: "IT" }
}) 	

10. Display total marks of students in each department
db.student.aggregate([
    {
        $group: {
            _id: "$dept",
            Total_Marks: { $sum: "$marks" }
        }
    }
])
